// Cargar Áreas Conocimiento (AreaConocimiento.csv)
LOAD CSV WITH HEADERS FROM 'file:///AreaConocimiento.csv' AS row
MERGE (a:AreaConocimiento {uri: row.uri})
SET a.nombre = row.foaf_name, a.codigo = row.aiiso_code;

// Cargar Titulaciones (AcademicDegree.csv)
LOAD CSV WITH HEADERS FROM 'file:///AcademicDegree.csv' AS row
MERGE (t:Titulacion {uri: row.uri})
SET t.nombre = row.foaf_name,
    t.codigo = row.aiiso_code,
    t.ou_cursoExtincion = row.ou_cursoExtincion, 
    t.ou_cursoInicioExtincion = row.ou_cursoInicioExtincion;

// Cargar Asignaturas (AsignaturaTitulacion.csv)
LOAD CSV WITH HEADERS FROM 'file:///AsignaturaTitulacion.csv' AS row
MERGE (asig:Asignatura {codigo: row.ou_asignatura_code})
SET asig.nombre = row.ou_asignatura_courseTitle,
    asig.creditos = row.ou_asignatura_ects

// Relación Asignatura -> Titulacion
